import abc
from typing import List, Dict, Any
from pydantic import BaseModel, Field

class ExternalProjectConfig(BaseModel):
    api_key: str = Field(..., description="API key for authentication")
    base_url: str = Field(..., description="Base URL of the external project API")
    auth_type: str = Field("token", description="Authentication type (e.g., 'token', 'oauth')")

class ExternalProjectInfo(BaseModel):
    id: str = Field(..., description="Unique identifier of the project in the external system")
    name: str = Field(..., description="Name of the external project")
    description: str | None = Field(None, description="Description of the external project")
    status: str = Field("active", description="Current status of the project")
    url: str | None = Field(None, description="URL to view the project in its native system")

class ExternalProjectInterface(abc.ABC):
    @abc.abstractmethod
    async def authenticate(self, config: ExternalProjectConfig) -> str:
        pass

    @abc.abstractmethod
    async def list_projects(self, auth_token: str) -> List[ExternalProjectInfo]:
        pass

    @abc.abstractmethod
    async def get_project_details(self, project_id: str, auth_token: str) -> ExternalProjectInfo:
        pass
